-- APK məlumatları cədvəli
create table if not exists public.apps (
  id bigint generated by default as identity primary key,
  name text not null,
  version text,
  category text,
  icon text default '📱',
  description text,
  file_path text not null,
  download_url text not null,
  created_at timestamptz not null default now()
);

alter table public.apps enable row level security;

-- Hamı tətbiqləri görə bilər
create policy "Public can read apps"
on public.apps for select
to anon, authenticated
using (true);

-- Admin siyahısı
create table if not exists public.admin_users (
  user_id uuid primary key references auth.users(id) on delete cascade
);

alter table public.admin_users enable row level security;

create policy "Admin can read own row"
on public.admin_users for select
to authenticated
using (user_id = auth.uid());

-- Yalnız admin tətbiq əlavə edə bilər
create policy "Admins insert apps"
on public.apps for insert
to authenticated
with check (
  exists (
    select 1 from public.admin_users
    where user_id = auth.uid()
  )
);

-- Yalnız admin tətbiq məlumatlarını dəyişə bilər
create policy "Admins update apps"
on public.apps for update
to authenticated
using (
  exists (
    select 1 from public.admin_users
    where user_id = auth.uid()
  )
)
with check (
  exists (
    select 1 from public.admin_users
    where user_id = auth.uid()
  )
);

-- Yalnız admin tətbiqi silə bilər
create policy "Admins delete apps"
on public.apps for delete
to authenticated
using (
  exists (
    select 1 from public.admin_users
    where user_id = auth.uid()
  )
);

-- APK yaddaş sahəsi
insert into storage.buckets
(id, name, public, file_size_limit, allowed_mime_types)
values (
  'apk-files',
  'apk-files',
  true,
  524288000,
  array[
    'application/vnd.android.package-archive',
    'application/octet-stream'
  ]
)
on conflict (id) do update
set public = true;

-- Admin APK yükləyə bilər
create policy "Admins upload APKs"
on storage.objects for insert
to authenticated
with check (
  bucket_id = 'apk-files'
  and exists (
    select 1 from public.admin_users
    where user_id = auth.uid()
  )
);

-- Admin APK silə bilər
create policy "Admins delete APKs"
on storage.objects for delete
to authenticated
using (
  bucket_id = 'apk-files'
  and exists (
    select 1 from public.admin_users
    where user_id = auth.uid()
  )
);
